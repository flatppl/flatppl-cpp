cmake_minimum_required(VERSION 3.26)
project(flatppl_cpp_tests LANGUAGES CXX)
enable_testing()

set(MODEL_NAME normal CACHE STRING "FlatPPL example to test")
set("${MODEL_NAME}_DIR" "${MODEL_DIR}")
find_package("${MODEL_NAME}" CONFIG REQUIRED NO_DEFAULT_PATH)
set(model_target "flatppl::${MODEL_NAME}")

foreach(mode linked open)
  add_executable("model_${mode}" "${MODEL_NAME}.cc")
  set_target_properties("model_${mode}" PROPERTIES CXX_EXTENSIONS OFF)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options("model_${mode}" PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()

target_compile_definitions(model_linked PRIVATE FLATPPL_TEST_LINKED)
target_link_libraries(model_linked PRIVATE "${model_target}")
target_link_libraries(model_open PRIVATE "${model_target}::headers")
add_test(NAME "${MODEL_NAME}_linked" COMMAND model_linked)
add_test(NAME "${MODEL_NAME}_open" COMMAND model_open "$<TARGET_FILE:${model_target}>")
if(WIN32)
  set_tests_properties("${MODEL_NAME}_linked" PROPERTIES
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:${model_target}>")
endif()
